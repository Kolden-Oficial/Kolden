## Story

Como desenvolvedor, quero ter o Supabase configurado com o schema completo do banco de dados, autenticação e storage, para que todos os dados do Omiron sejam persistidos com segurança e conformidade com LGPD desde o início.

---

## Acceptance Criteria

1. Projeto Supabase criado com todas as 12 tabelas do schema (conforme PRD FR-2, FR-3, FR-4)
2. Row Level Security (RLS) ativado e configurado em todas as tabelas com dados sensíveis
3. Políticas RLS impedem que um paciente acesse dados de outro paciente
4. Dr. Ariosto (role: `doctor`) tem acesso a dados de seus pacientes apenas
5. Secretária (role: `secretary`) vê apenas dados básicos dos pacientes (nome, email, foto)
6. Supabase Storage configurado com buckets `photos`, `pdfs`, `audio` e políticas de acesso corretas
7. Schema Prisma gerado e sincronizado com o banco (`npx prisma generate` sem erros)
8. Variáveis de ambiente documentadas em `.env.example`
9. Migrations versionadas em `supabase/migrations/`

## Dev Notes

### Schema SQL — 12 tabelas

```sql
-- Tabela base de usuários (complementa auth.users do Supabase)
CREATE TABLE users (
  id UUID REFERENCES auth.users(id) PRIMARY KEY,
  email TEXT NOT NULL,
  role TEXT NOT NULL CHECK (role IN ('patient', 'secretary', 'doctor')),
  name TEXT NOT NULL,
  photo_url TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- Perfil clínico do paciente
CREATE TABLE patient_profiles (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES users(id) NOT NULL,
  doctor_id UUID REFERENCES users(id),
  diagnoses TEXT[] DEFAULT '{}',
  medications JSONB DEFAULT '[]',
  goals TEXT[] DEFAULT '{}',
  notes TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 7 áreas de monitoramento (seed via migration)
CREATE TABLE monitoring_areas (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  name TEXT NOT NULL,
  icon TEXT NOT NULL,
  description TEXT,
  order_index INTEGER NOT NULL,
  active BOOLEAN DEFAULT TRUE
);

-- Check-ins diários
CREATE TABLE daily_checkins (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  patient_id UUID REFERENCES users(id) NOT NULL,
  area_id UUID REFERENCES monitoring_areas(id) NOT NULL,
  date DATE NOT NULL DEFAULT CURRENT_DATE,
  value INTEGER CHECK (value BETWEEN 1 AND 10),
  notes TEXT,
  photo_url TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  UNIQUE(patient_id, area_id, date)
);

-- Escalas diagnósticas
CREATE TABLE diagnostic_scales (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  patient_id UUID REFERENCES users(id) NOT NULL,
  scale_type TEXT NOT NULL CHECK (scale_type IN ('HAM-A', 'HAM-D', 'MADRS', 'YMRS', 'COCOCOLOS')),
  responses JSONB NOT NULL,
  score INTEGER,
  filled_at TIMESTAMPTZ DEFAULT NOW(),
  appointment_date DATE
);

-- Mensagens de chat
CREATE TABLE messages (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  sender_id UUID REFERENCES users(id) NOT NULL,
  receiver_id UUID REFERENCES users(id) NOT NULL,
  content TEXT,
  attachment_url TEXT,
  read_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Relatórios enviados pelo paciente
CREATE TABLE reports (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  patient_id UUID REFERENCES users(id) NOT NULL,
  content TEXT NOT NULL,
  attachments TEXT[] DEFAULT '{}',
  sent_at TIMESTAMPTZ DEFAULT NOW()
);

-- Gamificação (planta virtual)
CREATE TABLE gamification (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  patient_id UUID REFERENCES users(id) UNIQUE NOT NULL,
  streak_days INTEGER DEFAULT 0,
  plant_stage TEXT DEFAULT 'seed' CHECK (plant_stage IN ('seed', 'sprout', 'seedling', 'adult', 'tree')),
  last_activity_at TIMESTAMPTZ,
  total_days INTEGER DEFAULT 0
);

-- Frases de impacto (seed em E5)
CREATE TABLE impact_phrases (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  disorder_type TEXT NOT NULL,
  day_of_month INTEGER NOT NULL CHECK (day_of_month BETWEEN 1 AND 31),
  content TEXT NOT NULL,
  author TEXT NOT NULL,
  source TEXT
);

-- Meditações guiadas (seed em E5)
CREATE TABLE meditations (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  title TEXT NOT NULL,
  audio_url TEXT NOT NULL,
  duration_seconds INTEGER NOT NULL,
  category TEXT NOT NULL,
  disorder_types TEXT[] DEFAULT '{}',
  active BOOLEAN DEFAULT TRUE
);

-- Eventos do calendário
CREATE TABLE calendar_events (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  patient_id UUID REFERENCES users(id) NOT NULL,
  title TEXT NOT NULL,
  type TEXT NOT NULL CHECK (type IN ('appointment', 'task', 'milestone', 'reminder')),
  event_date DATE NOT NULL,
  notes TEXT,
  reminder_sent BOOLEAN DEFAULT FALSE
);

-- Notificações
CREATE TABLE notifications (
  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  user_id UUID REFERENCES users(id) NOT NULL,
  type TEXT NOT NULL,
  content TEXT NOT NULL,
  read BOOLEAN DEFAULT FALSE,
  scheduled_at TIMESTAMPTZ,
  created_at TIMESTAMPTZ DEFAULT NOW()
);
```

### Exemplo de política RLS

```sql
-- Habilitar RLS
ALTER TABLE patient_profiles ENABLE ROW LEVEL SECURITY;

-- Paciente vê apenas seu próprio perfil
CREATE POLICY "patient_sees_own_profile" ON patient_profiles
  FOR SELECT USING (user_id = auth.uid());

-- Médico vê perfis dos seus pacientes
CREATE POLICY "doctor_sees_patients" ON patient_profiles
  FOR SELECT USING (
    doctor_id = auth.uid() AND
    EXISTS (SELECT 1 FROM users WHERE id = auth.uid() AND role = 'doctor')
  );

-- Médico pode editar qualquer campo da ficha
CREATE POLICY "doctor_can_edit_patient" ON patient_profiles
  FOR UPDATE USING (
    doctor_id = auth.uid() AND
    EXISTS (SELECT 1 FROM users WHERE id = auth.uid() AND role = 'doctor')
  );
```
